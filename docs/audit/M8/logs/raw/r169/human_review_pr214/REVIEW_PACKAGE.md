def word_freq(text):
    """
    Return a dictionary mapping each word in the input text to its frequency.
    Words are considered as sequences separated by whitespace, transformed
    to lowercase, and stripped of the punctuation characters: . , ! ?

    Parameters
    ----------
    text : str
        Input string to analyze.

    Returns
    -------
    dict
        A dictionary where keys are lowercased words without the specified
        punctuation, and values are the counts of their occurrences.

    Notes
    -----
    - If *text* is not a string, an empty dictionary is returned.
    - Empty words resulting from punctuation removal are ignored.
    """
    if not isinstance(text, str):
        return {}

    # Punctuation characters to remove
    _punctuation = ".,!?"
    # Translation table to delete the punctuation characters
    trans = str.maketrans('', '', _punctuation)

    freq = {}
    for word in text.split():
        cleaned = word.translate(trans).lower()
        if cleaned:  # skip empty results
            freq[cleaned] = freq.get(cleaned, 0) + 1

    return freq
