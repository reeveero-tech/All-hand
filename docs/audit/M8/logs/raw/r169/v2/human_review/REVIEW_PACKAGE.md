def rle_encode(s: str) -> str:
    """
    Run‑length encode the input string s.

    Parameters
    ----------
    s : str
        The string to encode.

    Returns
    -------
    str
        The run‑length encoded representation. Empty string encodes to
        an empty string.

    Examples
    --------
    >>> rle_encode('aaabb')
    'a3b2'
    >>> rle_encode('')
    ''
    """
    if not s:
        return ''
    result = []
    count = 1
    for i in range(1, len(s)):
        if s[i] == s[i - 1]:
            count += 1
        else:
            result.append(f"{s[i - 1]}{count}")
            count = 1
    result.append(f"{s[-1]}{count}")
    return "".join(result)

    """Run‑length encode a string.

    by the count prefixed only when the count exceeds 1.  An empty string
